#include <iostream>
#include <cstdlib>
#include <ctime>
using namespace std;

int main() {
    srand(time(0));
    int num, guess, attempts;
    char play;

    do {
        num = rand() % 100 + 1;
        attempts = 0;

        cout << "\nGuess a number (1-100): ";

        do {
            cin >> guess;
            attempts++;

            if (guess > num)
                cout << "Too High! Try again: ";
            else if (guess < num)
                cout << "Too Low! Try again: ";
            else
                cout << "Correct! You won in " << attempts << " attempts!\n";

        } while (guess != num);

        cout << "Play again? (y/n): ";
        cin >> play;

    } while (play == 'y' || play == 'Y');

    cout << "Thanks for playing!";
    return 0;
}
